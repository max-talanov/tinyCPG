\subsection*{Supplementary Algorithms: experimental protocols}

\begin{algorithm}[h]
\caption{Initialisation robustness --- force-triggered medium walk over ten
$(\mu,\mathrm{CV})$ initialisations (\texttt{run\_cutforce\_sweep6.sh}).}\label{alg:main}
\begin{algorithmic}[1]
\Require Ten $(\mu_i,\mathrm{CV}_i)$ initialisation points; base seed $S_0=12345$
\Ensure Ten HDF5 files \texttt{cpg\_cutforce6\_robustness\_idx<i>\_*.h5}
\For{$i \in \{0, 1, \dots, 9\}$ \textbf{in parallel} (SLURM array task $i$)}
  \State $\text{seed} \gets S_0 + 10007 \cdot i$
  \State Initialise plastic weights (CUT$\to$RG-E, BS$\to$RG-E/F) $w \sim \mathrm{Lognormal}(\mu_i, \mathrm{CV}_i)$
  \State Configure medium walk (Table~\ref{tab:modes}), $\lambda=10^{-3}$, BS $=60$\,Hz tonic
  \State Start leg R in stance; both legs' CUT on for $T_{\text{lead}}$
  \For{each $100$\,ms step up to $120$\,s}
    \State Simulate $100$\,ms; update activation, fatigue (Eq.~\ref{eq:fatigue}), force, length, $r_{\text{Ia}}$ (Eq.~\ref{eq:ia})
    \For{each leg}
      \State Update running stance peak $\hat F$; switch stance$\leftrightarrow$swing on
             $F_E \le \theta_{\text{off}}\hat F$ / $F_E \ge \theta_{\text{on}}\hat F$ or at $T_{\max}$
      \State Set CUT and heel$\to$toe Ia-E (stance) or flexor-afferent (swing) rates
    \EndFor
    \State Log stance state, forces, population rates; sample STDP weights every $1$\,s
  \EndFor
  \State Write HDF5 with rates, weights, stance state and run attributes
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[h]
\caption{Locomotion-mode $\times$ learning-rate grid (\texttt{run\_forcetrig\_grid.sh}).}\label{alg:grid}
\begin{algorithmic}[1]
\Require $(\mu, \mathrm{CV}) = (3.5, 0.30)$, $\mathrm{seed} = 12345$
\Ensure Twenty HDF5 files \texttt{cpg\_ftgrid\_<mode>\_<lam>\_*.h5}
\For{$i \in \{0, 1, \dots, 19\}$ \textbf{in parallel} (SLURM array task $i$)}
  \State $\mathrm{mode} \gets \{\text{slow, medium, fast, toe}\}[\lfloor i / 5 \rfloor]$
  \State $\lambda \gets \{10^{-2},10^{-3},10^{-4},10^{-5},10^{-6}\}[i \bmod 5]$
  \State Configure the mode's operating point (Table~\ref{tab:modes}) and \verb|--stdp-lambda|$=\lambda$
  \State Simulate $120$\,s as in Algorithm~\ref{alg:main}
\EndFor
\State Report per cell over the last 20\,s: corr$(F_E,F_F)$, corr(RG-E,RG-F),
       corr$(F_{E,L},F_{E,R})$, fraction of stance bouts at $T_{\max}$, peak Force-E,
       learned CUT$\to$RG-E weight
\end{algorithmic}
\end{algorithm}
